Laboratorio: la maldición del ganador y la replicación

Laboratorio · se ejecuta en tu navegador

Si solo se publican los resultados significativos, los efectos publicados están inflados, sobre todo con muestras pequeñas. Elige el efecto real y el tamaño de muestra: el laboratorio simula 2000 estudios, «publica» los significativos y calcula qué pasaría al replicarlos.

rangoG = window.Lab.rango
viewof dReal = rangoG("Efecto real (d de Cohen)", [0.1, 0.8], {step: 0.05, value: 0.3, formato: (x) => x.toFixed(2).replace(".", ",")})
viewof nGrupo = rangoG("Participantes por grupo", [10, 200], {step: 5, value: 30})
{
  let semilla = 12345; const azar = () => { semilla |= 0; semilla = semilla + 0x6D2B79F5 | 0; let t = Math.imul(semilla ^ semilla >>> 15, 1 | semilla); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
  const normal = () => Math.sqrt(-2 * Math.log(1 - azar())) * Math.cos(2 * Math.PI * azar());
  const Phi = (z) => { const t = 1 / (1 + 0.2316419 * Math.abs(z)), d = 0.3989423 * Math.exp(-z * z / 2); const p = d * t * (0.3193815 + t * (-0.3565638 + t * (1.781478 + t * (-1.821256 + t * 1.330274)))); return z > 0 ? 1 - p : p; };
  const se = Math.sqrt(2 / nGrupo);
  const est = Array.from({length: 2000}, () => dReal + se * normal());
  const sig = est.filter((d) => d / se > 1.96);
  const potencia = sig.length / est.length;
  const dPub = sig.length ? sig.reduce((a, b) => a + b, 0) / sig.length : NaN;
  const potRep = 1 - Phi(1.96 - dReal / se);
  const nPlan = Math.ceil(2 * Math.pow((1.96 + 0.84) / dPub, 2));
  const potPlan = 1 - Phi(1.96 - dReal / Math.sqrt(2 / nPlan));
  const lo = -0.6, hi = 1.6, bins = 44, w = (hi - lo) / bins, cuenta = Array(bins).fill(0), cuentaS = Array(bins).fill(0);
  est.forEach((d) => { const i = Math.floor((d - lo) / w); if (i >= 0 && i < bins) { cuenta[i]++; if (d / se > 1.96) cuentaS[i]++; } });
  const mx = Math.max(...cuenta), W = 640, H = 220, x = (v) => 30 + (v - lo) / (hi - lo) * (W - 50), y = (c) => H - 30 - c / mx * (H - 50);
  let svg = `<svg viewBox="0 0 ${W} ${H}" width="100%" style="max-width:680px" role="img" aria-label="Histograma de 2000 efectos estimados; los significativos se muestran en color oscuro" font-family="inherit">`;
  cuenta.forEach((c, i) => { const x0 = x(lo + i * w); svg += `<rect x="${x0}" y="${y(c)}" width="${x(lo + (i + 1) * w) - x0 - 1}" height="${H - 30 - y(c)}" fill="#c9cfd6"/><rect x="${x0}" y="${y(cuentaS[i])}" width="${x(lo + (i + 1) * w) - x0 - 1}" height="${H - 30 - y(cuentaS[i])}" fill="#2a78d6"/>`; });
  svg += `<line x1="30" x2="${W - 20}" y1="${H - 30}" y2="${H - 30}" stroke="currentColor"/>`;
  [0, 0.5, 1, 1.5].forEach((v) => svg += `<text x="${x(v)}" y="${H - 12}" font-size="12" text-anchor="middle" fill="currentColor">${String(v).replace(".", ",")}</text>`);
  svg += `<line x1="${x(dReal)}" x2="${x(dReal)}" y1="10" y2="${H - 30}" stroke="#b3261e" stroke-width="2" stroke-dasharray="5 3"/><text x="${x(dReal) + 4}" y="22" font-size="12" fill="#b3261e">efecto real</text>`;
  if (!isNaN(dPub)) svg += `<line x1="${x(dPub)}" x2="${x(dPub)}" y1="30" y2="${H - 30}" stroke="#1c6b3a" stroke-width="2"/><text x="${x(dPub) + 4}" y="42" font-size="12" fill="#1c6b3a">media publicada</text>`;
  svg += "</svg>";
  const f = (v, k = 2) => v.toFixed(k).replace(".", ",");
  const d = html`<div role="status" aria-live="polite"></div>`;
  d.innerHTML = `${svg}<p class="small">Gris: todos los estudios. Azul: los significativos (p &lt; 0,05), que son los que se «publican».</p>
  <ul><li>Potencia del diseño: <strong>${Math.round(potencia * 100)} %</strong> de los estudios salen significativos.</li>
  <li>Efecto medio publicado: <strong>${isNaN(dPub) ? "—" : f(dPub)}</strong>, frente al real ${f(dReal)}: ${isNaN(dPub) ? "" : `<strong>${f(dPub / dReal, 1)} veces mayor</strong>`}.</li>
  <li>Una réplica con la misma muestra solo saldría significativa el <strong>${Math.round(potRep * 100)} %</strong> de las veces.</li>
  <li>Si planificas la réplica para tener un 80 % de potencia con el efecto publicado, usarías ${isNaN(nPlan) ? "—" : nPlan} por grupo, pero su potencia real sería del <strong>${Math.round(potPlan * 100)} %</strong>.</li></ul>
  <p>${potencia < 0.5 ? "Con poca potencia, lo publicado exagera el efecto y las réplicas «fracasan» aunque el efecto exista. Planifica las réplicas con un efecto más prudente (por ejemplo, la mitad del publicado) o con el menor efecto de interés." : "Con potencia alta, la inflación es pequeña: los resultados publicados se parecen al efecto real."}</p>`;
  return d;
}

Laboratorio: diagrama de flujo PRISMA 2020

Laboratorio · se ejecuta en tu navegador

Toda revisión sistemática debe informar del recorrido de los registros, desde la búsqueda hasta los estudios incluidos, con el diagrama de PRISMA 2020 (Page et al., 2021). Introduce tus números: el laboratorio comprueba que cuadran y dibuja el diagrama, que puedes descargar en SVG.

{
  const campos = [["bd", "Registros identificados en bases de datos", 1240], ["reg", "Registros de registros de ensayos u otras fuentes", 36], ["dup", "Duplicados eliminados", 310], ["auto", "Eliminados por otras razones antes del cribado", 0], ["excl1", "Excluidos por título y resumen", 820], ["norec", "Informes no recuperados", 6], ["excl2", "Informes excluidos tras lectura completa", 98], ["incl", "Estudios incluidos en la revisión", 42]];
  const raiz = html`<div class="prisma-lab"><div class="row g-2"></div></div>`;
  const ins = Object.fromEntries(campos.map(([id, et, v]) => { const d = html`<div class="col-sm-6"><label class="form-label small" for="pr-${id}">${et}</label><input id="pr-${id}" type="number" min="0" class="form-control form-control-sm" value="${v}"></div>`; raiz.firstElementChild.append(d); return [id, d.querySelector("input")]; }));
  const razones = html`<div class="mt-2"><label class="form-label small" for="pr-raz">Razones de exclusión tras lectura completa (una por línea, «razón: n»)</label><textarea id="pr-raz" class="form-control form-control-sm" rows="3">Diseño no elegible: 51\nPoblación distinta: 30\nSin datos del resultado: 17</textarea></div>`;
  const salida = html`<div role="status" aria-live="polite" class="mt-3"></div>`;
  raiz.append(razones, salida);
  const esc = (s) => String(s).replace(/[&<>]/g, (c) => ({"&": "&amp;", "<": "&lt;", ">": "&gt;"}[c]));
  function dibujar() {
    const v = Object.fromEntries(Object.entries(ins).map(([k, e]) => [k, Math.max(0, parseInt(e.value) || 0)]));
    const cribados = v.bd + v.reg - v.dup - v.auto, buscados = cribados - v.excl1, evaluados = buscados - v.norec, calc = evaluados - v.excl2;
    const raz = razones.querySelector("textarea").value.split("\n").map((l) => l.trim()).filter(Boolean);
    const sumaRaz = raz.reduce((a, l) => a + (parseInt((l.match(/(\d+)\s*$/) || [0, 0])[1]) || 0), 0);
    const avisos = [];
    if (cribados < 0 || buscados < 0 || evaluados < 0) avisos.push("Hay más exclusiones que registros en alguna fase: revisa los números.");
    if (calc !== v.incl) avisos.push(`Los números no cuadran: deberían quedar ${calc} estudios incluidos y has escrito ${v.incl}.`);
    if (raz.length && sumaRaz !== v.excl2) avisos.push(`Las razones de exclusión suman ${sumaRaz}, pero has excluido ${v.excl2} informes.`);
    const caja = (x, y, w, h, lineas, fondo = "#ffffff") => `<rect x="${x}" y="${y}" width="${w}" height="${h}" fill="${fondo}" stroke="#0b0b0b" stroke-width="1.2"/>` + lineas.map((l, i) => `<text x="${x + 8}" y="${y + 18 + i * 15}" font-size="12" fill="#0b0b0b">${esc(l)}</text>`).join("");
    const flecha = (x1, y1, x2, y2) => `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" stroke="#0b0b0b" stroke-width="1.2" marker-end="url(#pr-f)"/>`;
    const fase = (y, h, t) => `<rect x="0" y="${y}" width="28" height="${h}" fill="#cfe0f7"/><text transform="translate(18 ${y + h / 2}) rotate(-90)" font-size="12" font-weight="700" text-anchor="middle" fill="#0b0b0b">${t}</text>`;
    let s = `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 640 ${470 + Math.max(0, raz.length - 3) * 15}" width="100%" style="max-width:680px;background:#fff" role="img" aria-label="Diagrama PRISMA 2020: ${v.bd + v.reg} registros identificados, ${cribados} cribados, ${evaluados} informes evaluados, ${v.incl} estudios incluidos" font-family="Helvetica, Arial, sans-serif"><defs><marker id="pr-f" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="#0b0b0b"/></marker></defs><rect width="640" height="100%" fill="#fff"/>`;
    s += fase(10, 110, "Identificación") + fase(130, 250, "Cribado") + fase(390, 60, "Incluidos");
    s += caja(40, 10, 260, 66, [`Registros identificados:`, `  bases de datos (n = ${v.bd})`, `  registros y otras fuentes (n = ${v.reg})`]);
    s += caja(360, 10, 270, 66, [`Eliminados antes del cribado:`, `  duplicados (n = ${v.dup})`, `  otras razones (n = ${v.auto})`]) + flecha(300, 43, 358, 43);
    s += flecha(170, 76, 170, 138) + caja(40, 140, 260, 40, [`Registros cribados (n = ${cribados})`]) + caja(360, 140, 270, 40, [`Excluidos (n = ${v.excl1})`]) + flecha(300, 160, 358, 160);
    s += flecha(170, 180, 170, 208) + caja(40, 210, 260, 40, [`Informes buscados (n = ${buscados})`]) + caja(360, 210, 270, 40, [`No recuperados (n = ${v.norec})`]) + flecha(300, 230, 358, 230);
    s += flecha(170, 250, 170, 278) + caja(40, 280, 260, 40, [`Informes evaluados (n = ${evaluados})`]) + caja(360, 280, 270, 30 + Math.max(1, raz.length) * 15, [`Excluidos (n = ${v.excl2}):`, ...raz.map((r) => "  " + r)]) + flecha(300, 300, 358, 300);
    s += flecha(170, 320, 170, 398) + caja(40, 400, 260, 40, [`Estudios incluidos (n = ${v.incl})`], "#e2f0e5") + "</svg>";
    const url = URL.createObjectURL(new Blob([s], {type: "image/svg+xml"}));
    salida.innerHTML = `${avisos.length ? `<div class="alert alert-warning small">${avisos.map((a) => "⚠ " + esc(a)).join("<br>")}</div>` : '<p class="small">✓ Los números cuadran en todas las fases.</p>'}${s}<p class="mt-2"><a class="btn btn-outline-primary btn-sm" download="prisma-2020.svg" href="${url}">Descargar SVG</a></p>`;
  }
  raiz.addEventListener("input", dibujar); dibujar();
  return raiz;
}

El diagrama simplificado sigue la versión para revisiones nuevas que solo buscan en bases de datos y registros. Si tu revisión es una actualización o usa otras vías de búsqueda (citas, webs), usa la plantilla completa de PRISMA 2020.
